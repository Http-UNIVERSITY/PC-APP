        pstm.setString(4,txtpassword.getText());

        gender=txtrbmale.getText();

        if(txtrbfamle.isSelected())
        {
        gender=txtrbfamle.getText();
        }
        pstm.setString(6,gender);
        pstm.executeUpdate();
        
        }
        catch (Exception ex)
        {
         JOptionPane.showMessageDialog(null,ex.getMessage());
         JOptionPane.showMessageDialog(null,"Record has been inserted into database successfully....");
        }
    }
